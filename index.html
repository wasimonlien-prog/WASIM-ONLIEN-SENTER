function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 1. जब कस्टमर नई फाइल अपलोड करे if (data.fileData) { var decodedBlob = Utilities.newBlob( Utilities.base64Decode(data.fileData), data.mimeType, data.filename ); // फाइल सीधे आपके Google Drive में सेव होगी var folder = DriveApp.getRootFolder(); var file = folder.createFile(decodedBlob); var fileUrl = file.getUrl(); var currentDate = new Date().toLocaleString(); sheet.appendRow([data.filename, fileUrl, currentDate]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } // 2. जब एडमिन पैनल से फाइलें लोड करनी हों if (data.action === "fetch") { var rows = sheet.getDataRange().getValues(); var files = []; for (var i = 1; i < rows.length; i++) { if (rows[i][0] !== "") { files.push({ name: rows[i][0], url: rows[i][1], date: rows[i][2], id: rows[i][1] }); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", files: files })) .setMimeType(ContentService.MimeType.JSON); } // 3. जब एडमिन पैनल से फाइल डिलीट करनी हो if (data.action === "delete") { var targetUrl = data.fileId; var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { var fileUrl = rows[i][1]; if (fileUrl == targetUrl || String(fileUrl).trim() == String(targetUrl).trim()) { try { var fileId = extractDriveId(fileUrl); if (fileId) { DriveApp.getFileById(fileId).setTrashed(true); } } catch (err) {} sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function extractDriveId(url) { var match = String(url).match(/[-\w]{25,}/); return match ? match[0] : null; }
